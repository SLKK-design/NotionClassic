# Compile only the uECC files for 68020 to unlock native MULS.L/MULU.L.
# On SE/30 (68030) this makes uECC_shared_secret ~5x faster.
# The optimization-level and curve defines must be passed as -D flags so
# that uECC.c (compiled as a separate translation unit) sees the same
# settings as the #defines in uecc_alt.c — otherwise uECC.c defaults to
# uECC_OPTIMIZATION_LEVEL=0 (unoptimized binary double-and-add) and the
# handshake takes ~24 s instead of ~8 s.
set(UECC_FLAGS "-m68020 -O3 \
 -Dasm_mult=1 \
 -Dasm_square=1 \
 -Dasm_solinas_p256=1 \
 -DuECC_OPTIMIZATION_LEVEL=2 \
 -DuECC_SQUARE_FUNC=1 \
 -DuECC_WORD_SIZE=4 \
 -DuECC_SUPPORTS_secp256r1=1 \
 -DuECC_SUPPORTS_secp160r1=0 \
 -DuECC_SUPPORTS_secp192r1=0 \
 -DuECC_SUPPORTS_secp224r1=0 \
 -DuECC_SUPPORTS_secp256k1=0 \
 -DuECC_VLI_NATIVE_LITTLE_ENDIAN=0")
set_source_files_properties(src/core/net/uECC.c src/core/net/uecc_alt.c PROPERTIES COMPILE_FLAGS "${UECC_FLAGS}")
set_source_files_properties(src/core/net/vli_mult_comba_m68k.s src/core/net/vli_square_comba_m68k.s src/core/net/solinas_p256_m68k.s PROPERTIES COMPILE_FLAGS "-m68020")

ENDIF()

file(GLOB_RECURSE APP_SOURCES
    src/*.cpp
    src/*.c
)

    # Entropy for the TLS keys: Time Manager jitter (see the file's header)

# names and every src/res/*.pict (scripts/make_icon_res.py)

    OUTPUT ${ICONS_R}
    COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/scripts/make_icon_res.py
            ${CMAKE_CURRENT_SOURCE_DIR}/src/res ${CMAKE_CURRENT_BINARY_DIR}/res
    DEPENDS scripts/make_icon_res.py src/res/iconorder.txt
            ${RES_PICT_FILES} ${PAGE_ICON_FILES}
    COMMENT "Packing icons into the app"
)

        src/wdefshell.r
        ${ICONS_R}
)

)
